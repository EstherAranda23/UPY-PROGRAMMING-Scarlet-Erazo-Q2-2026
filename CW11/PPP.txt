# INPUT SECTION
# Create a dictionary to hold all the config variables
SET config ← EMPTY_DICTIONARY

# Load and read the configuration file
OPEN "config.txt" FOR READING AS my_file

FOR EACH line IN my_file
    # Clean up spaces around the text
    REMOVE_WHITESPACES(line)
    
    # Split parameter name and value by the equals sign
    SPLIT line BY "=" INTO param AND val
    
    # Check if value is float or int and convert it
    IF val CONTAINS "." THEN
        SET numeric_val ← TO_FLOAT(val)
    ELSE
        SET numeric_val ← TO_INT(val)
    ENDIF
    
    # Save it into the config dictionary
    SET config[param] ← numeric_val
ENDFOR

# Close the file after reading
CLOSE my_file

# Save config values into simple variables to use later
SET width ← config["ancho"]
SET height ← config["alto"]
SET max_iter ← config["max_iter"]

# Prepare the final CSV output file
OPEN "mandelbrot.csv" FOR WRITING AS output_csv
WRITE "row,column,iterations\n" TO output_csv


# PROCESS & OUTPUT SECTION
# Start looping through every single pixel on the screen
FOR EACH row FROM 0 TO height - 1
    FOR EACH column FROM 0 TO width - 1
        
        # PROCESS: Map the current pixel position to the complex plane coordinates
        SET real_part ← config["real_min"] + (column / width) * (config["real_max"] - config["real_min"])
        SET imag_part ← config["imag_min"] + (row / height) * (config["imag_max"] - config["imag_min"])
        
        # Define the complex number C based on coordinates
        SET c ← COMPLEX(real_part, imag_part)
        
        # Initialize Z at the starting position (0,0) and reset iterations counter
        SET z ← COMPLEX(0, 0)
        SET iterations ← 0
        
        # PROCESS: Mandelbrot math loop (Z = Z^2 + C)
        # Keep iterating until Z escapes or we hit max_iter limit
        WHILE (MAGNITUDE(z) <= 2) AND (iterations < max_iter)
            SET z ← (z * z) + c
            SET iterations ← iterations + 1
        ENDWHILE
        
        # OUTPUT: Save the row, column, and iteration number into the CSV file
        WRITE row + "," + column + "," + iterations + "\n" TO output_csv
        
    ENDFOR
ENDFOR

# Close the output file when completely done
CLOSE output_csv
